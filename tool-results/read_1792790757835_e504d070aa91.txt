    79→
    80→export function PresupuestosTable({
    81→  presupuestos,
    82→  loading,
    83→  editing,
    84→  configuracion,
    85→  onEdit,
    86→  onChanged,
    87→}: PresupuestosTableProps) {
    88→  const { toast } = useToast();
    89→  const [busqueda, setBusqueda] = useState("");
    90→  const [porEliminar, setPorEliminar] = useState<Presupuesto | null>(null);
    91→  const [enVista, setEnVista] = useState<Presupuesto | null>(null);
    92→  const [paraImprimir, setParaImprimir] = useState<Presupuesto | null>(null);
    93→  const [paraEnviar, setParaEnviar] = useState<Presupuesto | null>(null);
    94→  const [eliminando, setEliminando] = useState(false);
    95→
    96→  /*
    97→   * Mientras la vista previa de impresión está abierta: bloquea el
    98→   * scroll de fondo, marca el body para el CSS de impresión, ajusta
    99→   * el título del documento (nombre del PDF sugerido) y cierra con Esc.
   100→   */
   101→  useEffect(() => {
   102→    if (!paraImprimir) return;
   103→    const tituloAnterior = document.title;
   104→    const folio = `PTO-${paraImprimir.id.slice(-6).toUpperCase()}`;
   105→    document.body.classList.add("imprimiendo");
   106→    document.body.style.overflow = "hidden";
   107→    document.title = `Presupuesto ${folio} - ${paraImprimir.cliente.razonSocial}`;
   108→    const onKeyDown = (event: KeyboardEvent) => {
   109→      if (event.key === "Escape") setParaImprimir(null);
   110→    };
   111→    window.addEventListener("keydown", onKeyDown);
   112→    return () => {
   113→      document.body.classList.remove("imprimiendo");
   114→      document.body.style.overflow = "";
   115→      document.title = tituloAnterior;
   116→      window.removeEventListener("keydown", onKeyDown);
   117→    };
   118→  }, [paraImprimir]);
   119→
   120→  const filtrados = useMemo(() => {
   121→    const q = busqueda.trim().toLowerCase();

   123→    return presupuestos.filter((p) => {
   124→      const campos = [
   125→        p.cliente.razonSocial,
   126→        p.fecha,
   127→        formatearFechaISO(p.fecha),
   128→        p.observaciones ?? "",
   129→        ...p.items.map((item) => item.descripcion),
   130→      ];
   131→      return campos.some((campo) => campo.toLowerCase().includes(q));
   132→    });
   133→  }, [presupuestos, busqueda]);
   134→

   136→    if (!porEliminar) return;

   138→    try {
   139→      const response = await fetch(`/api/presupuestos/${porEliminar.id}`, {
   140→        method: "DELETE",
   141→      });

   143→        const data = await response.json().catch(() => ({}));
   144→        toast({
   145→          variant: "destructive",
   146→          title: "No se pudo eliminar",
   147→          description: data?.error ?? "Intente nuevamente.",
   148→        });
   149→        return;
   150→      }
   151→      toast({
   152→        title: "Presupuesto eliminado",
   153→        description: `El presupuesto de "${porEliminar.cliente.razonSocial}" fue eliminado.`,
   154→      });
   155→      setPorEliminar(null);
   156→      onChanged();
   157→    } catch {
   158→      toast({
   159→        variant: "destructive",
   160→        title: "Error de conexión",
   161→        description:
   162→          "No fue posible eliminar el presupuesto. Intente de nuevo.",
   163→      });
   164→    } finally {
   165→      setEliminando(false);
   166→    }
   167→  }
   168→
   169→  return (
   170→    <Card className="flex h-full flex-col">
   171→      <CardHeader>
   172→        <div className="flex flex-wrap items-start justify-between gap-3">
   173→          <div className="min-w-0">
   174→            <CardTitle className="text-lg">Presupuestos registrados</CardTitle>
   175→            <CardDescription>
   176→              {loading
   177→                ? "Cargando información…"
   178→                : `${presupuestos.length} ${
   179→                    presupuestos.length === 1 ? "presupuesto" : "presupuestos"
   180→                  } en total · ${filtrados.length} mostrados`}

   182→          </div>

   185→            size="icon"

   190→            <RefreshCw
   191→              className={loading ? "h-4 w-4 animate-spin" : "h-4 w-4"}
   192→              aria-hidden="true"
   193→            />
   194→          </Button>
   195→        </div>
   196→
   197→        <div className="relative mt-1">
   198→          <Search
   199→            className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground"
   200→            aria-hidden="true"
   201→          />
   202→          <Input
   203→            value={busqueda}
   204→            onChange={(event) => setBusqueda(event.target.value)}
   205→            placeholder="Buscar por cliente, fecha, producto…"
   206→            className="pl-9"
   207→            aria-label="Buscar presupuestos"
   208→          />
   209→        </div>
   210→      </CardHeader>
   211→
   212→      <CardContent className="flex-1 pt-0">
   213→        {loading ? (
   214→          <div
   215→            className="space-y-3"
   216→            aria-busy="true"
   217→            aria-label="Cargando presupuestos"
   218→          >
   219→            {Array.from({ length: 3 }).map((_, index) => (
   220→              <div key={index} className="flex items-center gap-4">
   221→                <Skeleton className="h-9 w-9 rounded-full" />
   222→                <div className="flex-1 space-y-2">
   223→                  <Skeleton className="h-4 w-1/2" />
   224→                  <Skeleton className="h-3 w-1/4" />
   225→                </div>
   226→                <Skeleton className="h-8 w-24" />
   227→              </div>
   228→            ))}
   229→          </div>
   230→        ) : presupuestos.length === 0 ? (
   231→          <div className="flex flex-col items-center justify-center rounded-lg border border-dashed py-14 text-center">
   232→            <ReceiptText
   233→              className="mb-3 h-10 w-10 text-muted-foreground/50"
   234→              aria-hidden="true"
   235→            />
   236→            <p className="font-medium">Aún no hay presupuestos registrados</p>

   238→              Use el formulario &quot;Registrar presupuesto&quot; para crear el
   239→              primero: elija el cliente, agregue productos y guarde.
   240→            </p>
   241→          </div>
   242→        ) : filtrados.length === 0 ? (
   243→          <div className="flex flex-col items-center justify-center rounded-lg border border-dashed py-14 text-center">
   244→            <Search
   245→              className="mb-3 h-8 w-8 text-muted-foreground/50"
   246→              aria-hidden="true"
   247→            />
   248→            <p className="font-medium">Sin resultados</p>
   249→            <p className="mt-1 text-sm text-muted-foreground">
   250→              Ningún presupuesto coincide con &quot;{busqueda}&quot;.
   251→            </p>
   252→          </div>

   254→          <TooltipProvider delayDuration={200}>
   255→            {/* Tarjetas (móvil) */}
   256→            <div className="space-y-3 sm:hidden">
   257→              {filtrados.map((presupuesto) => {
   258→                const esSeleccionado = editing?.id === presupuesto.id;
   259→                return (
   260→                  <div
   261→                    key={presupuesto.id}
   262→                    className={`rounded-lg border p-3 ${
   263→                      esSeleccionado ? "bg-accent/60" : undefined
   264→                    }`}
   265→                  >
   266→                    <div className="flex items-center justify-between gap-2">
   267→                      <span className="text-xs tabular-nums text-muted-foreground">
   268→                        {formatearFechaISO(presupuesto.fecha)}
   269→                      </span>
   270→                      <span className="text-sm font-semibold tabular-nums">
   271→                        {formatearMoneda(presupuesto.total)}
   272→                      </span>
   273→                    </div>
   274→                    <p
   275→                      className="mt-0.5 truncate font-medium"
   276→                      title={presupuesto.cliente.razonSocial}
   277→                    >
   278→                      {presupuesto.cliente.razonSocial}
   279→                    </p>
   280→                    <p className="text-xs text-muted-foreground">
   281→                      {presupuesto.items.length}{" "}
   282→                      {presupuesto.items.length === 1
   283→                        ? "producto"
   284→                        : "productos"}{" "}
   285→                      {presupuesto.items.length > 0 &&
   286→                        `· ${formatearMoneda(presupuesto.subtotal)}`}
   287→                    </p>
   288→                    <div className="mt-2 flex justify-end gap-1 border-t pt-2">
   289→                      <Tooltip>
   290→                        <TooltipTrigger asChild>
   291→                          <Button
   292→                            variant="ghost"
   293→                            size="icon"
   294→                            className="h-8 w-8"
   295→                            onClick={() => setEnVista(presupuesto)}
   296→                            aria-label={`Ver presupuesto de ${presupuesto.cliente.razonSocial}`}
   297→                          >
   298→                            <Eye className="h-4 w-4" aria-hidden="true" />
   299→                          </Button>
   300→                        </TooltipTrigger>
   301→                        <TooltipContent>Ver detalle</TooltipContent>
   302→                      </Tooltip>
   303→
   304→                      <Tooltip>
   305→                        <TooltipTrigger asChild>
   306→                          <Button
   307→                            variant="ghost"
   308→                            size="icon"
   309→                            className="h-8 w-8"
   310→                            onClick={() => setParaImprimir(presupuesto)}
   311→                            aria-label={`Imprimir presupuesto de ${presupuesto.cliente.razonSocial}`}
   312→                          >
   313→                            <Printer className="h-4 w-4" aria-hidden="true" />
   314→                          </Button>
   315→                        </TooltipTrigger>
   316→                        <TooltipContent>Imprimir / PDF</TooltipContent>
   317→                      </Tooltip>
   318→
   319→                      <Tooltip>
   320→                        <TooltipTrigger asChild>
   321→                          <Button
   322→                            variant="ghost"
   323→                            size="icon"
   324→                            className="h-8 w-8"
   325→                            onClick={() => setParaEnviar(presupuesto)}
   326→                            aria-label={`Enviar por correo el presupuesto de ${presupuesto.cliente.razonSocial}`}
   327→                          >
   328→                            <Mail className="h-4 w-4" aria-hidden="true" />
   329→                          </Button>
   330→                        </TooltipTrigger>
   331→                        <TooltipContent>Enviar por correo</TooltipContent>
   332→                      </Tooltip>
   333→
   334→                      <Tooltip>
   335→                        <TooltipTrigger asChild>
   336→                          <Button
   337→                            variant="ghost"
   338→                            size="icon"
   339→                            className="h-8 w-8"
   340→                            onClick={() => onEdit(presupuesto)}
   341→                            aria-label={`Editar presupuesto de ${presupuesto.cliente.razonSocial}`}
   342→                          >
   343→                            <Pencil className="h-4 w-4" aria-hidden="true" />
   344→                          </Button>
   345→                        </TooltipTrigger>
   346→                        <TooltipContent>Editar</TooltipContent>
   347→                      </Tooltip>
   348→
   349→                      <Tooltip>
   350→                        <TooltipTrigger asChild>
   351→                          <Button
   352→                            variant="ghost"
   353→                            size="icon"
   354→                            className="h-8 w-8 text-destructive hover:text-destructive"
   355→                            onClick={() => setPorEliminar(presupuesto)}
   356→                            aria-label={`Eliminar presupuesto de ${presupuesto.cliente.razonSocial}`}
   357→                          >
   358→                            <Trash2 className="h-4 w-4" aria-hidden="true" />
   359→                          </Button>
   360→                        </TooltipTrigger>
   361→                        <TooltipContent>Eliminar</TooltipContent>
   362→                      </Tooltip>
   363→                    </div>
   364→                  </div>
   365→                );
   366→              })}
   367→            </div>
   368→
   369→            {/* Tabla (tableta / escritorio) */}
   370→            <ScrollArea className="hidden max-h-[560px] pr-3 sm:block [&_[data-slot=table-container]]:overflow-x-visible">
   371→              <Table className="table-fixed">
   372→                <TableHeader className="sticky top-0 z-10 bg-card">
   373→                  <TableRow>
   374→                    <TableHead className="w-[11%] whitespace-nowrap">
   375→                      Fecha
   376→                    </TableHead>
   377→                    <TableHead className="w-[27%]">Cliente</TableHead>
   378→                    <TableHead className="hidden w-[15%] lg:table-cell">
   379→                      Productos
   380→                    </TableHead>
   381→                    <TableHead className="hidden w-[13%] whitespace-nowrap px-2 text-right sm:table-cell">
   382→                      Sub total
   383→                    </TableHead>
   384→                    <TableHead className="hidden w-[11%] whitespace-nowrap px-2 text-right sm:table-cell">
   385→                      IVA
   386→                    </TableHead>
   387→                    <TableHead className="w-[12%] whitespace-nowrap px-2 text-right">
   388→                      Total
   389→                    </TableHead>
   390→                    <TableHead className="w-[148px] text-right">
   391→                      Acciones
   392→                    </TableHead>
   393→                  </TableRow>
   394→                </TableHeader>
   395→                <TableBody>
   396→                  {filtrados.map((presupuesto) => {
   397→                    const esSeleccionado = editing?.id === presupuesto.id;
   398→                    const primerProducto =
   399→                      presupuesto.items[0]?.descripcion ?? "";
   400→                    const restantes = presupuesto.items.length - 1;
   401→                    return (
   402→                      <TableRow
   403→                        key={presupuesto.id}
   404→                        className={esSeleccionado ? "bg-accent/60" : undefined}
   405→                      >
   406→                        <TableCell className="whitespace-nowrap align-top text-sm tabular-nums">
   407→                          {formatearFechaISO(presupuesto.fecha)}
   408→                        </TableCell>
   409→
   410→                        <TableCell className="align-top">
   411→                          <span
   412→                            className="block truncate font-medium"
   413→                            title={presupuesto.cliente.razonSocial}
   414→                          >
   415→                            {presupuesto.cliente.razonSocial}
   416→                          </span>
   417→                          <p className="text-xs text-muted-foreground lg:hidden">
   418→                            {presupuesto.items.length}{" "}
   419→                            {presupuesto.items.length === 1
   420→                              ? "producto"
   421→                              : "productos"}
   422→                          </p>
   423→                        </TableCell>
   424→
   425→                        <TableCell className="hidden align-top lg:table-cell">
   426→                          <span
   427→                            className="block truncate text-sm"
   428→                            title={primerProducto}
   429→                          >
   430→                            {primerProducto}
   431→                          </span>
   432→                          {restantes > 0 && (
   433→                            <p className="truncate text-xs text-muted-foreground">
   434→                              + {restantes}{" "}
   435→                              {restantes === 1 ? "producto más" : "productos más"}
   436→                            </p>
   437→                          )}
   438→                        </TableCell>
   439→
   440→                        <TableCell className="hidden whitespace-nowrap px-2 align-top text-right text-sm tabular-nums sm:table-cell">
   441→                          {formatearMoneda(presupuesto.subtotal)}
   442→                        </TableCell>
   443→
   444→                        <TableCell className="hidden whitespace-nowrap px-2 align-top text-right text-sm tabular-nums text-muted-foreground sm:table-cell">
   445→                          {formatearMoneda(presupuesto.iva)}
   446→                        </TableCell>
   447→
   448→                        <TableCell className="whitespace-nowrap px-2 align-top text-right text-sm font-semibold tabular-nums">
   449→                          {formatearMoneda(presupuesto.total)}
   450→                        </TableCell>
   451→
   452→                        <TableCell className="text-right">
   453→                          <div className="flex justify-end gap-1">
   454→                            <Tooltip>
   455→                              <TooltipTrigger asChild>
   456→                                <Button
   457→                                  variant="ghost"
   458→                                  size="icon"
   459→                                  className="h-8 w-8"
   460→                                  onClick={() => setEnVista(presupuesto)}
   461→                                  aria-label={`Ver presupuesto de ${presupuesto.cliente.razonSocial}`}
   462→                                >
   463→                                  <Eye
   464→                                    className="h-4 w-4"
   465→                                    aria-hidden="true"
   466→                                  />
   467→                                </Button>
   468→                              </TooltipTrigger>
   469→                              <TooltipContent>Ver detalle</TooltipContent>
   470→                            </Tooltip>
   471→
   472→                            <Tooltip>
   473→                              <TooltipTrigger asChild>
   474→                                <Button
   475→                                  variant="ghost"
   476→                                  size="icon"
   477→                                  className="h-8 w-8"
   478→                                  onClick={() => setParaImprimir(presupuesto)}
   479→                                  aria-label={`Imprimir presupuesto de ${presupuesto.cliente.razonSocial}`}
   480→                                >
   481→                                  <Printer
   482→                                    className="h-4 w-4"
   483→                                    aria-hidden="true"
   484→                                  />
   485→                                </Button>
   486→                              </TooltipTrigger>
   487→                              <TooltipContent>Imprimir / PDF</TooltipContent>
   488→                            </Tooltip>
   489→
   490→                            <Tooltip>
   491→                              <TooltipTrigger asChild>
   492→                                <Button
   493→                                  variant="ghost"
   494→                                  size="icon"
   495→                                  className="h-8 w-8"
   496→                                  onClick={() => setParaEnviar(presupuesto)}
   497→                                  aria-label={`Enviar por correo el presupuesto de ${presupuesto.cliente.razonSocial}`}
   498→                                >
   499→                                  <Mail
   500→                                    className="h-4 w-4"
   501→                                    aria-hidden="true"
   502→                                  />
   503→                                </Button>
   504→                              </TooltipTrigger>
   505→                              <TooltipContent>Enviar por correo</TooltipContent>
   506→                            </Tooltip>
   507→
   508→                            <Tooltip>
   509→                              <TooltipTrigger asChild>
   510→                                <Button
   511→                                  variant="ghost"
   512→                                  size="icon"
   513→                                  className="h-8 w-8"
   514→                                  onClick={() => onEdit(presupuesto)}
   515→                                  aria-label={`Editar presupuesto de ${presupuesto.cliente.razonSocial}`}
   516→                                >
   517→                                  <Pencil
   518→                                    className="h-4 w-4"
   519→                                    aria-hidden="true"
   520→                                  />
   521→                                </Button>
   522→                              </TooltipTrigger>
   523→                              <TooltipContent>Editar</TooltipContent>
   524→                            </Tooltip>
   525→
   526→                            <Tooltip>
   527→                              <TooltipTrigger asChild>
   528→                                <Button
   529→                                  variant="ghost"
   530→                                  size="icon"
   531→                                  className="h-8 w-8 text-destructive hover:text-destructive"
   532→                                  onClick={() => setPorEliminar(presupuesto)}
   533→                                  aria-label={`Eliminar presupuesto de ${presupuesto.cliente.razonSocial}`}
   534→                                >
   535→                                  <Trash2
   536→                                    className="h-4 w-4"
   537→                                    aria-hidden="true"
   538→                                  />
   539→                                </Button>
   540→                              </TooltipTrigger>
   541→                              <TooltipContent>Eliminar</TooltipContent>
   542→                            </Tooltip>
   543→                          </div>
   544→                        </TableCell>
   545→                      </TableRow>
   546→                    );
   547→                  })}
   548→                </TableBody>
   549→              </Table>
   550→            </ScrollArea>

   553→      </CardContent>
   554→
   555→      {/* Detalle del presupuesto */}
   556→      <Dialog
   557→        open={enVista !== null}
   558→        onOpenChange={(open) => !open && setEnVista(null)}
   559→      >

   562→            <>
   563→              <DialogHeader>
   564→                <DialogTitle className="flex items-center gap-2">

   569→                  Presupuesto · {formatearFechaISO(enVista.fecha)}

   575→
   576→              <Button
   577→                variant="outline"

   589→              {enVista.observaciones && (

   615→                    </TableHeader>

   720→      </AlertDialog>
   721→
   722→      {/* Envío del presupuesto por correo electrónico */}
   723→      <EnviarCorreoDialog
   724→        presupuesto={paraEnviar}

   726→      />
   727→
   728→      {/* Vista previa de impresión del documento formal */}

   735→                size="sm"
   736→                onClick={() => setParaImprimir(null)}

   749→            </div>

   754→              />
   755→            </div>
   756→          </div>,
